// 「信息类」伪节点：机场用来展示剩余流量 / 到期时间的假节点，不是真节点。
// 它们如果进入 url-test / fallback / load-balance 组，会导致该组选中一个
// 连不通的"节点"从而整体失效，必须排除。
// 注意：原脚本用的关键词是「 剩余 」「 到期 」，前后带半角空格，
//       而机场实际用的名字是「剩余流量：1017.92 GB」（全角冒号、无空格），
//       所以原写法从未命中过。
// 如果你的机场有真实节点误命中，把对应关键词从下面删掉即可。
const INFO_NODE_REGEX =
  /(剩余|流量|到期|过期|重置|套餐|订阅|官网|网址|客服|公告|通知|机场|失联|Expire|Traffic|Website|Reset)/i;

function main(params) {
  if (!params || !Array.isArray(params.proxies)) return params;
  overwriteBasicOptions(params);
  applyClientFingerprint(params);
  overwriteSniffer(params);
  overwriteProxyGroups(params);
  overwriteRules(params);
  overwriteDns(params);
  return params;
}

// ---------------------------------------------------------------------------
// 客户端 TLS 指纹
// mihomo 已移除顶层 `global-client-fingerprint`，内核会直接报错：
//   "The `global-client-fingerprint` configuration is removed,
//    please set `client-fingerprint` directly on the proxy instead"
// 改为逐节点设置。
// ---------------------------------------------------------------------------
const CLIENT_FINGERPRINT = "chrome"; // 设为 null 表示不处理，沿用订阅自带的值
const FORCE_FINGERPRINT = false; // true = 强制覆盖订阅自带的 client-fingerprint
const FP_TYPES = ["vmess", "vless", "trojan", "hysteria", "hysteria2", "tuic", "anytls"];

function applyClientFingerprint(params) {
  if (!CLIENT_FINGERPRINT) return;
  for (const proxy of params.proxies) {
    if (!FP_TYPES.includes(proxy.type)) continue; // 不支持该字段的协议直接跳过
    if (!FORCE_FINGERPRINT && proxy["client-fingerprint"] !== undefined) continue;
    proxy["client-fingerprint"] = CLIENT_FINGERPRINT;
  }
}

// ---------------------------------------------------------------------------
// 保护区校验
// 只做告警、不抛异常 —— 抛异常会导致配置生成失败、直接断网。
// 改动脚本后若在日志里看到 "[保护区]" 开头的告警，说明定制规则集被动了。
// ---------------------------------------------------------------------------
function verifyProtected(params) {
  const missing = [];

  const providers = params["rule-providers"] || {};
  for (const name of PROTECTED_PROVIDER_NAMES) {
    if (!providers[name]) missing.push("rule-provider: " + name);
  }

  const rules = params.rules || [];
  for (const rule of PROTECTED_RULE_ENTRIES) {
    if (rules.indexOf(rule) === -1) missing.push("rule: " + rule);
  }

  if (missing.length > 0 && typeof console !== "undefined" && typeof console.warn === "function") {
    console.warn("[保护区] 定制规则集缺失，请检查脚本改动 -> " + missing.join(" | "));
  }

  return missing;
}
